# Siri Integration Phase 2 — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship phase-2 Siri integration: `INPlayMediaIntent` extension (`AbbayIntents`) + App Group (`group.com.edyuto.abbay`) + graceful launch-state flag with retry-on-play recovery + live disambiguation + `INPlayMediaIntent` donations.

**Architecture:** New `AbbayIntents` app-extension conforms to `INPlayMediaIntentHandling`, returns `.handleInApp` so iOS background-launches the host app for actual playback. App Group `UserDefaults` carries three keys: `abbay.lastPlayedAudioURL`, `abbay.librarySnapshot.v1` (JSON of `[{title, urlString, bookmarkData?}]`), `abbay.audioSessionUnavailable`. `AppDelegate.fatalError` becomes `LaunchStateFlag.markUnavailable()`. `Player.play(_:)` retries `setActive` at the top to recover the flag on every play. Donations fire after every successful play.

**Tech Stack:** Swift 5, iOS 17+, `Intents` framework (legacy SiriKit, NOT AppIntents — phase 1 already covers that), AVFoundation, XCTest, xcodegen.

**Reference spec:** `docs/superpowers/specs/2026-05-17-siri-phase2-design.md` (merged in #78). Read it before starting.

```swift
import Foundation

enum LastPlayedStore {
    // Public so AppDelegateSeeder, IntentRouter, and extension can reference it
    static let key: String = "abbay.lastPlayedAudioURL"

    static var defaults: UserDefaults = AppGroup.defaults
    static var standardFallback: UserDefaults = .standard

    static func save(url: URL?) {
        guard let url = url else {
            defaults.removeObject(forKey: key)
            return
        }
        // Seed-then-write: if AppGroup empty AND .standard has value, copy first
        if defaults.string(forKey: key) == nil,
           let standardValue = standardFallback.string(forKey: key) {
            defaults.set(standardValue, forKey: key)
        }
        defaults.set(url.absoluteString, forKey: key)
    }

    static func lastPlayedURL() -> URL? {
        if let value = defaults.string(forKey: key) {
            return URL(string: value)
        }
        if let standardValue = standardFallback.string(forKey: key) {
            return URL(string: standardValue)
        }
        return nil
    }
}
```

import Foundation

struct LibrarySnapshotEntry: Codable, Equatable {
    let title: String
    let urlString: String
    let bookmarkData: Data?
}

struct LibrarySnapshot: Codable, Equatable {
    let version: Int
    let updatedAt: Date
    let audios: [LibrarySnapshotEntry]
}

protocol LibrarySnapshotWriterProtocol {
    func write(audios: [AudioModel])
    func refresh()
}

final class LibrarySnapshotWriter: LibrarySnapshotWriterProtocol {
    static let key: String = "abbay.librarySnapshot.v1"

    static var defaults: UserDefaults = AppGroup.defaults
    static var shared: LibrarySnapshotWriterProtocol = LibrarySnapshotWriter()

    private let loadAllAudios: LoadAllAudiosUseCaseProtocol

    init(loadAllAudios: LoadAllAudiosUseCaseProtocol = LoadAllAudiosUseCase()) {
        self.loadAllAudios = loadAllAudios
    }

    func write(audios: [AudioModel]) {
        let entries = audios.map { audio in
            LibrarySnapshotEntry(
                title: audio.title,
                urlString: audio.url?.absoluteString ?? "",
                bookmarkData: (audio.bookmarkData?.isEmpty == false) ? audio.bookmarkData : nil
            )
        }
        let snapshot = LibrarySnapshot(version: 1, updatedAt: Date(), audios: entries)
        if let data = try? JSONEncoder().encode(snapshot) {
            Self.defaults.set(data, forKey: Self.key)
        }
    }

    func refresh() {
        // Implemented in Task 6
        fatalError("refresh() implemented in Task 6")
    }
}
```

protocol IntentDonorProtocol {

            artwork: nil

            mediaItems: [mediaItem],
            mediaContainer: nil,
            playShuffled: nil,
            playbackRepeatMode: .none,
            resumePlayback: nil,
            playbackQueueLocation: .now,
            playbackSpeed: nil,

                mediaType: .audioBook,

                mediaName: model.title,
                artistName: nil,
                albumName: nil,
                genreNames: nil,

import Intents

protocol IntentRouterProtocol {
    func handle(_ intent: INPlayMediaIntent) -> INPlayMediaIntentResponse
}

final class IntentRouter: IntentRouterProtocol {
    private let loadAudio: LoadAudioUseCaseProtocol
    private let snapshotWriter: LibrarySnapshotWriterProtocol
    private let snapshotReader: () -> [LibrarySnapshotEntry]
    private let player: () -> PlayerProtocol

    init(
        loadAudio: LoadAudioUseCaseProtocol = LoadAudioUseCase(),
        snapshotWriter: LibrarySnapshotWriterProtocol = LibrarySnapshotWriter.shared,
        snapshotReader: @escaping () -> [LibrarySnapshotEntry] = { LibrarySnapshotReader.read() },
        player: @escaping () -> PlayerProtocol = { DependencyContainer.shared.player }
    ) {
        self.loadAudio = loadAudio
        self.snapshotWriter = snapshotWriter
        self.snapshotReader = snapshotReader
        self.player = player
    }

    func handle(_ intent: INPlayMediaIntent) -> INPlayMediaIntentResponse {
        if LaunchStateFlag.isUnavailable() {
            return INPlayMediaIntentResponse(code: .failureUnknownMediaType, userActivity: nil)
        }
        guard let identifier = intent.mediaItems?.first?.identifier else {
            return INPlayMediaIntentResponse(code: .failureUnknownMediaType, userActivity: nil)
        }

        let resolvedURL = resolveURL(from: identifier)
        guard var model = loadAudio.execute(url: resolvedURL) else {
            if let saved = LastPlayedStore.lastPlayedURL(), saved.absoluteString == identifier {
                LastPlayedStore.save(url: nil)
            }
            snapshotWriter.refresh()
            return INPlayMediaIntentResponse(code: .failureUnknownMediaType, userActivity: nil)
        }

        player().play(&model)
        return INPlayMediaIntentResponse(code: .success, userActivity: nil)
    }

    private func resolveURL(from identifier: String) -> URL? {
        let entries = snapshotReader()
        if let match = entries.first(where: { $0.urlString == identifier }),
           let bookmark = match.bookmarkData {
            var isStale = false
            if let resolved = try? URL(
                resolvingBookmarkData: bookmark,
                options: .withoutUI,
                relativeTo: nil,
                bookmarkDataIsStale: &isStale
            ) {
                return resolved
            }
        }
        return URL(string: identifier)
    }
}

import Foundation

// Returns the single top match (used by PlayBookByNameIntent phase-1 path)
func fuzzyMatch(input: String, audios: [AudioModel], lastPlayedURL: URL?) -> AudioModel? {
    matchAll(input: input, audios: audios, lastPlayedURL: lastPlayedURL).first
}

// Returns all matches at the top score (used by extension PlayMediaIntentHandler disambiguation path)
func matchAll(input: String, audios: [AudioModel], lastPlayedURL: URL?) -> [AudioModel] {
    let normalizedInput = normalize(input)
    guard !normalizedInput.isEmpty else { return [] }

    let scored: [(AudioModel, Int, Int)] = audios.enumerated().compactMap { idx, audio in
        let s = score(audio: audio, normalizedInput: normalizedInput)
        return s > 0 ? (audio, s, idx) : nil
    }

    guard let topScore = scored.map(\.1).max() else { return [] }

    let topMatches = scored.filter { $0.1 == topScore }
    return topMatches.sorted { lhs, rhs in
        let lhsIsLast = lastPlayedURL.map { lhs.0.url == $0 } ?? false
        let rhsIsLast = lastPlayedURL.map { rhs.0.url == $0 } ?? false
        if lhsIsLast != rhsIsLast { return lhsIsLast }
        return lhs.2 < rhs.2
    }.map { $0.0 }
}

private func normalize(_ s: String) -> String {
    var v = s.replacingOccurrences(of: "&", with: " and ")
    v = v.lowercased()
    v = v.components(separatedBy: CharacterSet.punctuationCharacters.union(.symbols)).joined()
    v = v.components(separatedBy: .whitespacesAndNewlines).filter { !$0.isEmpty }.joined(separator: " ")
    return v
}

private func score(audio: AudioModel, normalizedInput: String) -> Int {
    let normalizedTitle = normalize(audio.title)
    if normalizedTitle == normalizedInput { return 3 }
    if normalizedTitle.hasPrefix(normalizedInput) { return 2 }
    if normalizedTitle.contains(normalizedInput) { return 1 }
    return 0
}
```

import Intents

final class PlayMediaIntentHandler: NSObject, INPlayMediaIntentHandling {
    func resolveMediaItems(
        for intent: INPlayMediaIntent,
        with completion: @escaping ([INPlayMediaMediaItemResolutionResult]) -> Void
    ) {
        if LaunchStateFlag.isUnavailable() {
            completion([.unsupported(forReason: .unsupportedMediaType)])
            return
        }

        let entries = LibrarySnapshotReader.read()
        guard !entries.isEmpty else {
            completion([.unsupported(forReason: .unsupportedMediaType)])
            return
        }

        let mediaName = intent.mediaSearch?.mediaName ?? ""
        let lastPlayedString = AppGroup.defaults.string(forKey: LastPlayedStore.key)
        let lastPlayedURL = lastPlayedString.flatMap { URL(string: $0) }

        let audios: [AudioModel] = entries.compactMap { entry in
            guard let url = URL(string: entry.urlString) else { return nil }
            var model = AudioModel(url: url)
            model.title = entry.title
            return model
        }

        let candidates = matchAll(input: mediaName, audios: audios, lastPlayedURL: lastPlayedURL)
        if candidates.isEmpty {
            completion([.unsupported(forReason: .unsupportedMediaType)])
            return
        }

        let items: [INMediaItem] = candidates.map { audio in
            INMediaItem(
                identifier: audio.url?.absoluteString ?? "",
                title: audio.title,
                type: .audioBook,
                artwork: nil
            )
        }

        if items.count == 1 {
            completion([.success(with: items[0])])
        } else if items.count <= 5 {
            completion([.disambiguation(with: items)])
        } else {
            completion([.success(with: items[0])])  // fallback to single-best
        }
    }

    func handle(
        intent: INPlayMediaIntent,
        completion: @escaping (INPlayMediaIntentResponse) -> Void
    ) {
        if LaunchStateFlag.isUnavailable() {
            // Use custom Response from .intentdefinition when bindings are available:
            // completion(PlayMediaIntentResponse.unavailable(/* per Xcode-generated case */))
            // For now, plain failure code:
            completion(INPlayMediaIntentResponse(code: .failureUnknownMediaType, userActivity: nil))
            return
        }

        // Defense-in-depth: per SiriKit lifecycle, when resolveMediaItems returns all-unsupported
        // (no-match path), Siri normally does NOT call handle — it speaks the failure dialog tied
        // to the Resolution Result + .intentdefinition Response template (set up in T21) directly.
        // This branch catches the unlikely case where handle is invoked with empty mediaItems.
        if intent.mediaItems?.isEmpty ?? true {
            completion(INPlayMediaIntentResponse(code: .failureUnknownMediaType, userActivity: nil))
            return
        }

        completion(INPlayMediaIntentResponse(code: .handleInApp, userActivity: nil))
    }
}
